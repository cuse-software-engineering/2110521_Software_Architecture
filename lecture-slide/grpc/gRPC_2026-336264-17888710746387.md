# Part 2 — Protocol Buffer

## Slide 7 — Protocol Buffer (.proto file)

https://developers.google.com/protocol-buffers/docs/overview

Interface Definition Language (IDL) for describing both the service interface and the structure of the payload messages.

```proto
message SearchRequest {
  required string query = 1;
  optional int32 page_number = 2;
  optional int32 result_per_page = 3;
}
```

- **Field numbers:** unique number, from 1 to 2^29 - 1 (536,870,911), except 19000 through 19999
- **Field rules:** `required` [1], `optional` [0..1], `repeated` [0..*]
- **Scalar types:** double, float, bool, string, bytes, int32/int64, uint32/uint64, sint32/sint64

## Slide 8 — Sample Protocol Buffer Message

```proto
message Person {
  string name = 1;
  int32 id = 2;
  bool has_ponycopter = 3;
}
```

`protoc` compiles the message into a `Person` class with fields `name`, `id`, `has_ponycopter` and generated `getter()`, `setter()`, `serialize()` methods.

## Slide 9 — Define gRPC services (.proto)

```proto
// The greeter service definition.
service Greeter {
  // Sends a greeting
  rpc SayHello (HelloRequest) returns (HelloResponse) {}
}

// The request message containing the user's name.
message HelloRequest {
  string greeting = 1;
}

// The response message containing the greetings
message HelloResponse {
  string reply = 1;
}
```

The **protoc compiler** generates both **client code** and **server code** from the `.proto` file.

## Slide 10 — 4 kinds of gRPC Service Method

| # | Kind | Client's Request | Server's Response |
|---|---|---|---|
| 1 | Unary RPCs | Single | Single |
| 2 | Server streaming RPCs | Single | Stream |
| 3 | Client streaming RPCs | Stream | Single |
| 4 | Bidirectional streaming RPCs | Stream | Stream |

*gRPC guarantees message order.

## Slide 23 — 1. Unary RPC Server Service

```js
function checkFeature(point) {
  var feature;
  // Check if there is already a feature object for the given point
  for (var i = 0; i < feature_list.length; i++) {
    feature = feature_list[i];
    if (feature.location.latitude === point.latitude &&
        feature.location.longitude === point.longitude) {
      return feature;
    }
  }
  var name = '';
  feature = {
    name: name,
    location: point
  };
  return feature;
}

function getFeature(call, callback) {
  callback(null, checkFeature(call.request));
}
```

## Slide 24 — 2. Server Streaming RPC Server Service

```js
function listFeatures(call) {
  var lo = call.request.lo;
  var hi = call.request.hi;
  var left = _.min([lo.longitude, hi.longitude]);
  var right = _.max([lo.longitude, hi.longitude]);
  var top = _.max([lo.latitude, hi.latitude]);
  var bottom = _.min([lo.latitude, hi.latitude]);
  // For each feature, check if it is in the given bounding box
  _.each(feature_list, function(feature) {
    if (feature.name === '') {
      return;
    }
    if (feature.location.longitude >= left &&
        feature.location.longitude <= right &&
        feature.location.latitude >= bottom &&
        feature.location.latitude <= top) {
      call.write(feature);   // send all messages
    }
  });
  call.end();
}
```

## Slide 25 — 3. Client Streaming RPC Server Service

```js
function recordRoute(call, callback) {
  var point_count = 0;
  var feature_count = 0;
  var distance = 0;
  var previous = null;
  var start_time = process.hrtime();
  call.on('data', function(point) {
    point_count += 1;
    if (checkFeature(point).name !== '') {
      feature_count += 1;
    }
    if (previous != null) {
      distance += getDistance(previous, point);
    }
    previous = point;
  });
  call.on('end', function() {
    callback(null, {
      point_count: point_count,
      feature_count: feature_count,
      // Cast the distance to an integer
      distance: distance|0,
      // End the timer
      elapsed_time: process.hrtime(start_time)[0]
    });
  });
}
```

Pattern:

```js
call.on('data', function(point) {
  // Process user data
});
call.on('end', function() {
  callback(null, result);
});
```

## Slide 26 — 4. Bidirectional Streaming RPC Server Service

```js
function routeChat(call) {
  call.on('data', function(note) {
    var key = pointKey(note.location);
    if (route_notes.hasOwnProperty(key)) {
      _.each(route_notes[key], function(note) {
        call.write(note);
      });
    } else {
      route_notes[key] = [];
    }
    route_notes[key].push(JSON.parse(JSON.stringify(note)));
  });
  call.on('end', function() {
    call.end();
  });
}
```

        return;

## Slide 34 — 3. Client Streaming RPC (client, 2 of 2)

```js
    function pointSender(lat, lng) {
      return function(callback) {
        console.log('Visiting point ' + lat/COORD_FACTOR + ', ' + lng/COORD_FACTOR);
        call.write({            // send lat,lng to the server through `call`
          latitude: lat,
          longitude: lng
        });
        _.delay(callback, _.random(500, 1500));
      };
    }
    var point_senders = [];
    for (var i = 0; i < num_points; i++) {
      var rand_point = feature_list[_.random(0, feature_list.length - 1)];
      point_senders[i] = pointSender(rand_point.location.latitude,
                                     rand_point.location.longitude);
    }
    async.series(point_senders, function() {   // call pointSender to send points to the server
      call.end();
    });
  });
}
```

## Slide 35 — 4. Bidirectional Streaming RPC (client)

```js
function runRouteChat(callback) {
  var call = client.routeChat();               // call routeChat() on the server
  call.on('data', function(note) {
    console.log('Got message "' + note.message + '" at ' +
      note.location.latitude + ', ' + note.location.longitude);
  });
  call.on('end', callback);                    // when the server finishes sending, run callback
  var notes = [
    { location: { latitude: 0, longitude: 0 }, message: 'First message' },
    { location: { latitude: 0, longitude: 1 }, message: 'Second message' },
    { location: { latitude: 1, longitude: 0 }, message: 'Third message' },
    { location: { latitude: 0, longitude: 0 }, message: 'Fourth message' }
  ];
  for (var i = 0; i < notes.length; i++) {
    var note = notes[i];
    console.log('Sending message "' + note.message + '" at ' +
      note.location.latitude + ', ' + note.location.longitude);
    call.write(note);                          // send data to the server through `call`
  }
  call.end();                                  // finish sending to the server
}
```
